-- =========================================================================
-- 1. DOMÍNIO GEOGRÁFICO E ELEITORAL (Tabelas Dimensão/Base)
-- =========================================================================

CREATE TABLE uf (
    sigla_uf VARCHAR(2) PRIMARY KEY,
    nome_uf VARCHAR(50)
);

CREATE TABLE municipio (
    id_municipio VARCHAR(10) PRIMARY KEY, -- IBGE 7 dígitos
    id_municipio_tse VARCHAR(10),
    sigla_uf VARCHAR(2) NOT NULL,
    FOREIGN KEY (sigla_uf) REFERENCES uf(sigla_uf)
);

CREATE TABLE zona_eleitoral (
    id_zona SERIAL PRIMARY KEY,
    zona INT NOT NULL,
    id_municipio VARCHAR(10) NOT NULL,
    FOREIGN KEY (id_municipio) REFERENCES municipio(id_municipio)
);

CREATE TABLE eleicao (
    id_eleicao VARCHAR(20) PRIMARY KEY,
    ano INT NOT NULL,
    tipo_eleicao VARCHAR(100),
    data_eleicao DATE,
    turno INT
);

CREATE TABLE partido (
    numero_partido VARCHAR(10) PRIMARY KEY,
    sigla VARCHAR(20),
    nome VARCHAR(150)
);

CREATE TABLE vaga (
    id_vaga SERIAL PRIMARY KEY,
    id_eleicao VARCHAR(20) NOT NULL,
    id_municipio VARCHAR(10), -- Pode ser NULL se a vaga for estadual/federal
    cargo VARCHAR(100) NOT NULL,
    vagas INT,
    FOREIGN KEY (id_eleicao) REFERENCES eleicao(id_eleicao),
    FOREIGN KEY (id_municipio) REFERENCES municipio(id_municipio)
);


-- =========================================================================
-- 2. DOMÍNIO DO CANDIDATO
-- =========================================================================

CREATE TABLE pessoa (
    titulo_eleitoral VARCHAR(20) PRIMARY KEY,
    cpf VARCHAR(20),
    nome VARCHAR(255) NOT NULL,
    data_nascimento DATE,
    genero VARCHAR(50),
    --instrucao VARCHAR(100),
    raca VARCHAR(50),
    --estado_civil VARCHAR(50),
    --nacionalidade VARCHAR(100),
    --ocupacao VARCHAR(150)
);

CREATE TABLE candidatura (
    sequencial_candidato VARCHAR(50) PRIMARY KEY,
    titulo_eleitoral VARCHAR(20) NOT NULL,
    id_eleicao VARCHAR(20) NOT NULL,
    numero_partido VARCHAR(10) NOT NULL,
    id_municipio VARCHAR(10),
    cargo VARCHAR(100) NOT NULL,
    numero_urna VARCHAR(20),
    nome_urna VARCHAR(255),
    situacao VARCHAR(100),
    FOREIGN KEY (titulo_eleitoral) REFERENCES pessoa(titulo_eleitoral),
    FOREIGN KEY (id_eleicao) REFERENCES eleicao(id_eleicao),
    FOREIGN KEY (numero_partido) REFERENCES partido(numero_partido),
    FOREIGN KEY (id_municipio) REFERENCES municipio(id_municipio)
);

CREATE TABLE bem_patrimonial (
    id_bem SERIAL PRIMARY KEY,
    sequencial_candidato VARCHAR(50) NOT NULL,
    tipo_item VARCHAR(150),
    descricao_item TEXT,
    valor_item DECIMAL(15, 2), -- Substitui o FLOAT64 para precisão monetária
    FOREIGN KEY (sequencial_candidato) REFERENCES candidatura(sequencial_candidato)
);


-- =========================================================================
-- 3. DOMÍNIO FINANCEIRO (Prestações de Contas)
-- =========================================================================

CREATE TABLE cnae (
    cnae_2_subclasse VARCHAR(20) PRIMARY KEY,
    cnae_2_classe VARCHAR(20),
    descricao_cnae VARCHAR(255)
);

CREATE TABLE agente_financeiro (
    cpf_cnpj VARCHAR(20) PRIMARY KEY,
    nome VARCHAR(255),
    nome_rf VARCHAR(255),
    cnae_2_subclasse VARCHAR(20),
    FOREIGN KEY (cnae_2_subclasse) REFERENCES cnae(cnae_2_subclasse)
);

CREATE TABLE receita (
    sequencial_receita VARCHAR(100) PRIMARY KEY,
    sequencial_candidato VARCHAR(50) NOT NULL,
    cpf_cnpj_doador VARCHAR(20),
    data_receita DATE,
    valor_receita DECIMAL(15, 2),
    origem_receita VARCHAR(150),
    natureza_receita VARCHAR(150),
    especie_receita VARCHAR(150),
    numero_documento VARCHAR(100),
    FOREIGN KEY (sequencial_candidato) REFERENCES candidatura(sequencial_candidato),
    FOREIGN KEY (cpf_cnpj_doador) REFERENCES agente_financeiro(cpf_cnpj)
);

CREATE TABLE despesa (
    sequencial_despesa VARCHAR(100) PRIMARY KEY,
    sequencial_candidato VARCHAR(50) NOT NULL,
    cpf_cnpj_fornecedor VARCHAR(20),
    data_despesa DATE,
    valor_despesa DECIMAL(15, 2),
    tipo_despesa VARCHAR(150),
    descricao_despesa TEXT,
    origem_despesa VARCHAR(150),
    FOREIGN KEY (sequencial_candidato) REFERENCES candidatura(sequencial_candidato),
    FOREIGN KEY (cpf_cnpj_fornecedor) REFERENCES agente_financeiro(cpf_cnpj)
);
